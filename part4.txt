#include <iostream>
#include <cmath>
#include <cstdlib>
#include <ctime>

int main() {
    // Buffon's Needle simulation parameters
    const double t = 20;
    const double len_needle = 10;
    const int num_needles = 10000000;

    // Initialize counters
    int total = 0;
    int intersecting = 0;

    // Buffon's Needle simulation
    srand(time(nullptr));

    for (int i = 0; i < num_needles; ++i) {
        double angle = ((double)rand() / RAND_MAX) * M_PI;
        double x = ((double)rand() / RAND_MAX) * 600;
        double y = ((double)rand() / RAND_MAX) * 600;

        if (std::abs(sin(angle) * len_needle / 2) > fmod(y, t)) {
            intersecting++;
            // No plotting in C++
        }

        total++;
    }

    // Estimate pi
    double probability = static_cast<double>(intersecting) / total;
    double pi_estimate = len_needle / (probability * t);

    // Print the estimated value of pi
    std::cout << "Estimated value of pi: " << std::fixed << pi_estimate << std::endl;

    return 0;
}
